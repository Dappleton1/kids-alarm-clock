display:
  - platform: lcd_pcf8574
    id: lcd

    update_interval: 1s
    lambda: |-
      auto now = id(ha_time).now();
      if (!now.is_valid()) { it.print(0, 0, "Syncing time..."); return; }
      it.strftime(0, 0, "%l:%M %p    %a", now);
      if (millis() < id(vol_shown_until)) { it.printf(0, 1, "Volume %2d of 30 ", (int) id(volume).state); return; }
      if (millis() < id(track_shown_until)) { it.printf(0, 1, "%-16.16s", id(alarm_tune).current_option().c_str()); return; }
      int st = id(alarm_state);
      if (st == 1) { it.print(0, 1, "** WAKE UP! **  "); return; }
      if (st == 2) { it.printf(0, 1, "Snooze %d min   ", (int) id(snooze_minutes).state); return; }
      if (!id(alarm_enabled).state) { it.print(0, 1, "Alarm off       "); return; }
      // Next ring, not today's setting: walk forward from now, weekend time on Sat/Sun or in holiday
      // mode, and let Skip Next swallow the first one. Tag = "on" if it is today, else the day name.
      static const char *DOW[] = {"", "Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"};
      int d0 = now.day_of_week, nowmin = now.hour * 60 + now.minute;
      bool skip = id(skip_next).state;
      int h = 0, m = 0, off = -1;
      for (int i = 0; i < 8 && off < 0; i++) {
        int d = ((d0 - 1 + i) % 7) + 1;
        bool weekend = (d == 1 || d == 7 || id(holiday_mode).state);
        int ah = weekend ? id(weekend_alarm).hour : id(weekday_alarm).hour;
        int am = weekend ? id(weekend_alarm).minute : id(weekday_alarm).minute;
        if (i == 0 && ah * 60 + am <= nowmin) continue;
        if (skip) { skip = false; continue; }
        h = ah; m = am; off = i;
      }
      const char *tag = (off == 0) ? (id(holiday_mode).state ? "hol" : "on") : DOW[((d0 - 1 + off) % 7) + 1];
      it.printf(0, 1, "Alarm %2d:%02d%s %-3s", (h % 12 == 0) ? 12 : h % 12, m, (h < 12) ? "a" : "p", tag);

# ---------- alarm logic ----------
script:
  # Fired by the datetime on_time triggers. Only ring if today matches the schedule.
  - id: alarm_due
    parameters:
      for_weekend: bool
    then:
      - if:
          condition:
            lambda: |-
              int d = id(ha_time).now().day_of_week;
              bool weekend = (d == 1 || d == 7 || id(holiday_mode).state);
              return id(alarm_enabled).state && id(alarm_state) == 0 && weekend == for_weekend;
          then:
            - script.execute: alarm_fire

  - id: alarm_fire
    then:
      - if:
          condition:
            switch.is_on: skip_next
          then:
            - switch.turn_off: skip_next
            - logger.log: "Skip Next was on - alarm skipped"
          else:
            - script.execute: start_ringing

  - id: start_ringing
    mode: restart
    then:
      - lambda: id(alarm_state) = 1;
      - text_sensor.template.publish: { id: alarm_state_text, state: "ringing" }
      - switch.turn_on: backlight
      - lambda: id(mp3).set_volume((uint8_t) id(volume).state);
      - delay: 100ms                       # DFPlayer needs a gap between commands
      - script.execute: play_tune
      - delay: 150ms
      - script.execute: { id: mp3_repeat, enable: true }   # module loops the track on its own
      - script.execute: ring_timeout

  # Pad 2: step to the next ringer and play 5 s of it at the set volume. Restart mode = rapid
  # presses keep stepping and the stop timer restarts. Does nothing while the alarm is live.
  - id: preview_track
    mode: restart
    then:
      - if:
          condition:
            lambda: return id(alarm_state) == 0;
          then:
            - select.next:
                id: alarm_tune
                cycle: true
            - lambda: id(track_shown_until) = millis() + 5000;
            - lambda: id(mp3).set_volume((uint8_t) id(volume).state);
            - delay: 100ms
            - script.execute: play_tune
            - delay: 5s
            - lambda: id(mp3).stop();

  # DFPlayer command 0x19 = repeat the track that is playing (arg 0 = on, 1 = off). ESPHome doesn't
  # expose it, so the frame is built by hand (same layout as the component's send_cmd_). The alarm
  # used to rely on the module's 'track finished' message to replay; one missed message = silence.
  - id: mp3_repeat
    parameters:
      enable: bool
    then:
      - lambda: |-
          uint8_t f[10] = {0x7E, 0xFF, 0x06, 0x19, 0x01, 0x00, (uint8_t) (enable ? 0x00 : 0x01), 0x00, 0x00, 0xEF};
          uint16_t sum = 0;
          for (int i = 1; i < 7; i++) sum += f[i];
          sum = -sum;
          f[7] = sum >> 8; f[8] = (uint8_t) sum;
          id(uart_mp3).write_array(f, 10);

  # selector position N -> /01/0NN.mp3 on the microSD
  - id: play_tune
    then:
      - lambda: |-
          auto idx = id(alarm_tune).active_index();
          int n = idx.has_value() ? (int) idx.value() + 1 : 1;
          id(mp3).play_folder(1, n);

  # Auto-stop so a slept-through alarm doesn't play all morning.
  - id: ring_timeout
    mode: restart
    then:
      - delay: !lambda return (uint32_t) (id(max_ring_minutes).state * 60000);
      - logger.log: "Alarm timed out - auto-dismiss"
      - script.execute: dismiss

  - id: snooze
    mode: restart
    then:
      - if:
          condition:
            lambda: return id(alarm_state) == 1;
          then:
            - script.execute: { id: mp3_repeat, enable: false }
            - delay: 100ms
            - lambda: id(mp3).stop();
            - script.stop: ring_timeout
            - lambda: id(alarm_state) = 2;
            - text_sensor.template.publish: { id: alarm_state_text, state: "snoozed" }
            - script.execute: backlight_timeout
            - delay: !lambda return (uint32_t) (id(snooze_minutes).state * 60000);
            - script.execute: start_ringing

  - id: dismiss
    then:
      - script.stop: snooze
      - script.stop: ring_timeout
      - script.execute: { id: mp3_repeat, enable: false }
      - delay: 100ms
      - lambda: id(mp3).stop();
      - lambda: id(alarm_state) = 0;
      - text_sensor.template.publish: { id: alarm_state_text, state: "idle" }
      - script.execute: backlight_timeout

  # ---------- backlight schedule ----------
  # Sets bl_quiet from the Lights Off / Lights On window (handles a window that crosses midnight).
  - id: backlight_eval
    then:
      - lambda: |-
          auto now = id(ha_time).now();
          if (!now.is_valid()) { id(bl_quiet) = false; return; }
          int cur = now.hour * 60 + now.minute;
          int off = id(lights_off_at).hour * 60 + id(lights_off_at).minute;
          int on  = id(lights_on_at).hour * 60 + id(lights_on_at).minute;
          id(bl_quiet) = (off <= on) ? (cur >= off && cur < on) : (cur >= off || cur < on);

  # Runs at time sync and at both window edges: enforce the schedule.
  - id: backlight_apply
    then:
      - script.execute: backlight_eval
      - if:
          condition:
            switch.is_on: auto_backlight
          then:
            - if:
                condition:
                  lambda: return id(bl_quiet) && id(alarm_state) != 1;
                then:
                  - switch.turn_off: backlight
                else:
                  - switch.turn_on: backlight

  # Any press: light up now; the timeout script decides whether it goes dark again.
  - id: backlight_wake
    then:
      - switch.turn_on: backlight

  # After the timeout, go dark only if inside the window, auto is on, and nothing is ringing.
  - id: backlight_timeout
    mode: restart
    then:
      - delay: !lambda return (uint32_t) (id(backlight_timeout_s).state * 1000);
      - script.execute: backlight_eval
      - if:
          condition:
            lambda: return id(auto_backlight).state && id(bl_quiet) && id(alarm_state) != 1;
          then:
            - switch.turn_off: backlight

  - id: boot_init
    then:
      - if:
          condition:
            switch.is_on: backlight
          then:
            - lambda: id(lcd).backlight();
          else:
            - lambda: id(lcd).no_backlight();
      - text_sensor.template.publish: { id: alarm_state_text, state: "idle" }
      - delay: 2s                          # DFPlayer takes ~1.5 s to read the card
      - lambda: id(mp3).set_volume((uint8_t) id(volume).state);
